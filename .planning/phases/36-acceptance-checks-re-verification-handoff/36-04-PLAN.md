h = base.joinpath("36-HANDOFF.md").read_text()
cd = base.joinpath("36-CODE-DIVERGENCE.md").read_text()
led = pathlib.Path(".planning/phases/32-evidence-framework-issue-ledger-conflict-adjudication/32-ISSUE-LEDGER.md").read_text()

# Partition 36-HANDOFF.md into the three classification tables by header, then
# require each finding ID to appear in EXACTLY ONE partition (exactly-once, not substring presence).
def partition(text, table_headers):
    parts = {}
    # split on markdown table/section headers
    chunks = re.split(r'\n(?=##\s)', text)
    for hdr in table_headers:
        blk = "\n".join(c for c in chunks if re.search(r'##\s*' + re.escape(hdr), c))
        parts[hdr] = blk
    return parts

tables = ["Fixed", "Intentionally Unchanged", "Deferred"]
parts = partition(h, tables)
def count_in_block(fid, blk):
    # match the finding ID as a leading cell in a table row (| fid | ...) to avoid substring false-positives
    return len(re.findall(r'(?m)^\|\s*' + re.escape(fid) + r'\s*\|', blk))

violations = []
for f in findings:
    counts = {t: count_in_block(f, parts[t]) for t in tables}
    total = sum(counts.values())

        violations.append(f"{f}: classified {total}x across tables {counts}")

    for t, c in counts.items():
        if c > 1:
            violations.append(f"{f}: appears {c}x inside the {t} table")
assert not violations, "handoff classification is not an exactly-once partition:\n  " + "\n  ".join(violations)
for sect in ["Fixed","Intentionally Unchanged","Deferred","FUT-01","FUT-04"]:
